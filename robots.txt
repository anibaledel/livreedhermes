# Choix explicite sur les robots d'IA (2026-09-21) : la recherche ET
# l'entraînement sont autorisés. Ils l'étaient déjà de fait, par le groupe
# « User-agent: * » ci-dessous — mais un fichier muet ne distingue pas une
# décision d'un oubli. Nommer chaque robot rend le choix lisible par un
# humain qui ouvre ce fichier comme par l'exploitant qui le lit.
#

# ATTENTION AU FONCTIONNEMENT DES GROUPES — un robot nommé ci-dessous lit
# SON groupe et ignore entièrement « User-agent: * ». Les Disallow doivent
# donc être répétés dans chaque groupe, sinon les robots nommés y auraient
# accès alors que tous les autres ne l'ont pas. Les dossiers de motifs
# (motifs-svg, motifs-pinterest, pro-downloads) ne sont plus interdits
# depuis le 2026-10-01 : tout y est en libre téléchargement, rien à cacher.
# Toute règle ajoutée au groupe « * » est à reporter partout ici.
#
# /pagefind/ : l'index de la recherche interne (fragments binaires lus par
# recherche.html dans le navigateur) — rien à explorer pour un robot.

# ── Tous les robots ────────────────────────────────────────────────────

Disallow: /includes/
